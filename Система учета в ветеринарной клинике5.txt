DROP DATABASE IF EXISTS vet_clinic;
CREATE DATABASE vet_clinic CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE vet_clinic;

1. Справочник видов животных
CREATE TABLE Species (
    species_id INT AUTO_INCREMENT PRIMARY KEY,
    species_name VARCHAR(50) NOT NULL
);
2. Справочник пород
CREATE TABLE Breeds (
    breed_id INT AUTO_INCREMENT PRIMARY KEY,
    breed_name VARCHAR(100) NOT NULL,
    species_id INT NOT NULL,
    CONSTRAINT fk_breeds_species FOREIGN KEY (species_id) REFERENCES Species(species_id) ON DELETE CASCADE
);
3. Владельцы животных
CREATE TABLE Owners (
    owner_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    phone VARCHAR(20) NOT NULL,
    address VARCHAR(255) NOT NULL
);
4. Животные
CREATE TABLE Animals (
    animal_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL,
    species_id INT NOT NULL,
    breed_id INT,
    birth_date DATE NOT NULL,
    gender CHAR(1) NOT NULL CHECK (gender IN ('М', 'Ж')),
    color VARCHAR(50),
    chip_number VARCHAR(30) UNIQUE,
    owner_id INT NOT NULL,
    CONSTRAINT fk_animals_species FOREIGN KEY (species_id) REFERENCES Species(species_id),
    CONSTRAINT fk_animals_breeds FOREIGN KEY (breed_id) REFERENCES Breeds(breed_id),
    CONSTRAINT fk_animals_owners FOREIGN KEY (owner_id) REFERENCES Owners(owner_id) ON DELETE CASCADE
);
5. Ветеринарные врачи
CREATE TABLE Vets (
    vet_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    specialization VARCHAR(100) NOT NULL
);
6. Приёмы животных
CREATE TABLE Appointments (
    appointment_id INT AUTO_INCREMENT PRIMARY KEY,
    appointment_datetime DATETIME NOT NULL,
    animal_id INT NOT NULL,
    vet_id INT NOT NULL,
    complaints TEXT,
    diagnosis TEXT NOT NULL,
    treatment TEXT,
    CONSTRAINT fk_appointments_animals FOREIGN KEY (animal_id) REFERENCES Animals(animal_id) ON DELETE CASCADE,
    CONSTRAINT fk_appointments_vets FOREIGN KEY (vet_id) REFERENCES Vets(vet_id)
);
7. Справочник групп препаратов
CREATE TABLE Medication_Groups (
    group_id INT AUTO_INCREMENT PRIMARY KEY,
    group_name VARCHAR(100) NOT NULL
);
8. Справочник стран
CREATE TABLE Countries (
    country_id INT AUTO_INCREMENT PRIMARY KEY,
    country_name VARCHAR(100) NOT NULL
);
9. Лекарственные препараты
CREATE TABLE Medications (
    medication_id INT AUTO_INCREMENT PRIMARY KEY,
    medication_name VARCHAR(150) NOT NULL,
    form VARCHAR(50) NOT NULL,
    dosage VARCHAR(50) NOT NULL,
    group_id INT NOT NULL,
    country_id INT NOT NULL,
    CONSTRAINT fk_medications_groups FOREIGN KEY (group_id) REFERENCES Medication_Groups(group_id),
    CONSTRAINT fk_medications_countries FOREIGN KEY (country_id) REFERENCES Countries(country_id)
);
10. Назначения (связующая таблица)
CREATE TABLE Prescriptions (
    appointment_id INT NOT NULL,
    medication_id INT NOT NULL,
    quantity INT NOT NULL CHECK (quantity > 0),
    price DECIMAL(10,2) NOT NULL CHECK (price >= 0),
    PRIMARY KEY (appointment_id, medication_id),
    CONSTRAINT fk_prescriptions_appointments FOREIGN KEY (appointment_id) REFERENCES Appointments(appointment_id) ON DELETE CASCADE,
    CONSTRAINT fk_prescriptions_medications FOREIGN KEY (medication_id) REFERENCES Medications(medication_id)
);

1. Where/Order by
SELECT 
    a.name AS Кличка, 
    s.species_name AS Вид, 
    b.breed_name AS Порода, 
    o.full_name AS Владелец
FROM Animals a, Species s, Owners o, Breeds b
WHERE a.species_id = s.species_id 
  AND a.owner_id = o.owner_id 
  AND a.breed_id = b.breed_id
ORDER BY o.full_name ASC;
2. Join
SELECT 
    ap.appointment_datetime AS Дата_приема, 
    an.name AS Животное, 
    v.full_name AS Врач, 
    ap.diagnosis AS Диагноз
FROM Appointments ap
INNER JOIN Animals an ON ap.animal_id = an.animal_id
INNER JOIN Vets v ON ap.vet_id = v.vet_id
ORDER BY ap.appointment_datetime DESC;
3. SUM, AVG, COUNT, MAX, MIN
SELECT 
    COUNT(*) AS Всего_назначений,        
    SUM(quantity) AS Всего_упаковок,     
    AVG(price) AS Средняя_цена,          
    MAX(price) AS Максимальная_цена,     
    MIN(price) AS Минимальная_цена       
FROM Prescriptions;
4. Group by/ having
SELECT 
    v.full_name AS Врач,
    COUNT(ap.appointment_id) AS Количество_приемов,
    SUM(pr.price * pr.quantity) AS Общая_сумма_назначений
FROM Vets v
JOIN Appointments ap ON v.vet_id = ap.vet_id
JOIN Prescriptions pr ON ap.appointment_id = pr.appointment_id
GROUP BY v.vet_id, v.full_name
HAVING COUNT(ap.appointment_id) >= 1; 
5.  IN, ALL
5.1. Использование IN (владельцы, у которых есть собаки)
SELECT full_name AS Владелец
FROM Owners
WHERE owner_id IN (
    SELECT owner_id 
    FROM Animals 
    WHERE species_id = (SELECT species_id FROM Species WHERE species_name = 'Собака')
);

5.2. Использование ALL (животные, которые старше всех кошек в базе)
SELECT name AS Кличка, birth_date AS Дата_рождения
FROM Animals
WHERE birth_date < ALL (
    SELECT birth_date 
    FROM Animals 
    WHERE species_id = (SELECT species_id FROM Species WHERE species_name = 'Кошка')
);

6. IF, CASE
6.1. Использование CASE (возрастные категории)
SELECT 
    name AS Кличка, 
    birth_date AS Дата_рождения,
    CASE 
        WHEN birth_date > '2022-01-01' THEN 'Молодое'
        WHEN birth_date > '2019-01-01' THEN 'Взрослое'
        ELSE 'Пожилое'
    END AS Возрастная_категория
FROM Animals;

6.2. Использование IF (скидка на препараты)
SELECT 
    appointment_id AS Код_приема,
    medication_id AS Код_препарата,
    quantity AS Количество,
    price AS Базовая_цена,
    IF(quantity > 2, price * quantity * 0.9, price * quantity) AS Итоговая_стоимость
FROM Prescriptions;

7.2. Временное представление (WITH / CTE)
Считаем, сколько каждый владелец потратил на лекарства
WITH OwnerSpending AS (
    SELECT 
        o.full_name AS Владелец, 
        SUM(p.price * p.quantity) AS Сумма_трат
    FROM Owners o
    JOIN Animals a ON o.owner_id = a.owner_id
    JOIN Appointments ap ON a.animal_id = ap.animal_id
    JOIN Prescriptions p ON ap.appointment_id = p.appointment_id
    GROUP BY o.full_name
)
SELECT * FROM OwnerSpending ORDER BY Сумма_трат DESC;
